### 4.2 Signatures

```java
// Resume.java
@Data @TableName("resume")
public class Resume implements Serializable {
    @TableId(type = IdType.AUTO)
    private Long id;
    private String slug;           // 短链接标识
    private String title;          // 简历标题
    private Long userId;           // 用户 ID
    private String data;           // JSON: 完整简历结构数据
    private String visibility;     // private/public
    private Boolean locked;        // 是否锁定
    private String source;         // ORIGINAL/AI_GENERATED
    private Long sourceResumeId;   // 来源简历 ID
    private String language;       // CHINESE/ENGLISH
    private Long templateId;       // 关联模板 ID
    private Date createdAt;
    private Date updatedAt;
}

// ResumeMapper.java
@Mapper
public interface ResumeMapper extends BaseMapper<Resume> {}

// ResumeService.java
@Service
public class ResumeService extends ServiceImpl<ResumeMapper, Resume> {
    public List<Resume> listByUserId(Long userId);
    public Resume createResume(Resume resume);
    public boolean updateResume(Resume resume);
    public boolean deleteResume(Long id);
}

// ResumeController.java
@RestController @RequestMapping("/api/resume")
public class ResumeController {
    @GetMapping("/list")           List<Resume> listByUserId(@RequestParam Long userId);
    @GetMapping("/{id}")           Resume getById(@PathVariable Long id);
    @PostMapping("/create")       Resume create(@RequestBody Resume resume);
    @PutMapping("/update")        boolean update(@RequestBody Resume resume);
    @DeleteMapping("/{id}")       boolean delete(@PathVariable Long id);
}
```
